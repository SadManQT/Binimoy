# LP-001 Bootstrap Reproducible Workspace Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A new checkout installs with pinned tools, starts PostgreSQL, and serves an explicitly labeled empty fixture shell with a running worker heartbeat.

**Architecture:** pnpm TypeScript monorepo laid out as spec A3. `apps/web` is a Next.js server app, `apps/worker` is a separate Node process, both read validated config from `packages/contracts` and reach PostgreSQL through `packages/db`. The existing FastAPI/React code moves to `legacy/` as read-only reference.

**Tech Stack:** Node 24, pnpm (via corepack), TypeScript, Next.js, PostgreSQL 17 (Docker), `pg`, zod, Vitest, Biome, tsx.

**Spec:** `docs/MASTER_SPEC.md` (currently `Lenden_Prohori_Master_Build_Specification_v7.md`; Task 1 moves it). Roadmap: `docs/superpowers/plans/2026-10-12-v7-takeover-roadmap.md`.

## Global Constraints

- No floating `latest` in committed manifests. Install with `-E` (exact) and commit `pnpm-lock.yaml`. (A2)
- Record the exact resolved versions in `docs/adr/ADR-001-runtime-and-deployment-shape.md`. Do not write a version number that a command did not print. (A2, E2)
- Non-secret defaults: `PAYMENT_MODE=mock`, `RESEARCH_MODE=fixture`, `AI_MODE=fixture`, `VERIFIER_MODE=mock`; `APP_ENV` is explicit with no default. (A12)
- A live mode with a missing key fails with an actionable error. It never falls back to fixtures. (A12)
- `.env.example` holds placeholders only. No secrets, `node_modules`, or build output are committed. (A7, A12)
- No seeded privileged users and no reset endpoints. (A7)
- Do not add scripts that do nothing. A11 scripts that do not exist yet are listed as absent in `STATUS.md`.
- New code never imports from `legacy/`.
- Work on branch `lp-001-bootstrap`. `main` is not touched.
- `pnpm setup` is a pnpm built-in. The project script must be run as `pnpm run setup`.
- End every commit message with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- Skeleton packages with an empty `index.ts` are created because A3 names them as required output of this task.

## Review Focus

1. An `.env` copied from `.env.example` has `SEARCH_API_KEY=` (empty string). Fixture mode must still start; empty means unset. Pinned in Task 3.
2. `RESEARCH_MODE=live` with no key must stop startup and name the variable. Pinned in Task 3.
3. Two processes migrate at once (web replica plus a migration job). Each migration must apply exactly once. Pinned in Task 4.
4. PostgreSQL is down. `/api/health` must report not ready with HTTP 503, not a healthy 200. Pinned in Task 6.
5. The inherited `.gitignore` ignores `.env.*`, which silently drops `.env.example` from the commit. Pinned in Task 1 with `git check-ignore`.

`packages/contracts/src/config.ts`:

```ts
import { z } from "zod";

const mode = z.enum(["fixture", "live"]).default("fixture");

const schema = z
  .object({
    APP_ENV: z.enum(["local", "test", "staging", "production"]),
    DATABASE_URL: z.string().min(1),
    PAYMENT_MODE: z.literal("mock").default("mock"),
    VERIFIER_MODE: z.literal("mock").default("mock"),
    RESEARCH_MODE: mode,
    AI_MODE: mode,
    SEARCH_API_KEY: z.string().optional(),
    MODEL_API_KEY: z.string().optional(),
  })
  .superRefine((c, ctx) => {
    if (c.RESEARCH_MODE === "live" && !c.SEARCH_API_KEY) {
      ctx.addIssue({
        code: "custom",
        path: ["SEARCH_API_KEY"],
        message: "RESEARCH_MODE=live needs SEARCH_API_KEY. Set it, or set RESEARCH_MODE=fixture.",
      });
    }
    if (c.AI_MODE === "live" && !c.MODEL_API_KEY) {
      ctx.addIssue({
        code: "custom",
        path: ["MODEL_API_KEY"],
        message: "AI_MODE=live needs MODEL_API_KEY. Set it, or set AI_MODE=fixture.",
      });
    }
  });

export type Config = z.infer<typeof schema>;

export function loadConfig(env: Record<string, string | undefined>): Config {
  // An .env line like `SEARCH_API_KEY=` yields "", which means unset.
  const set = Object.fromEntries(Object.entries(env).filter(([, v]) => v !== ""));
  const result = schema.safeParse(set);
  if (!result.success) {
    // Report variable names only, never values.
    const names = [...new Set(result.error.issues.map((i) => i.path.join(".")))];
    const custom = result.error.issues.filter((i) => i.code === "custom").map((i) => i.message);
    throw new Error(
      `Invalid configuration: ${names.join(", ")} missing or not allowed. ${custom.join(" ")}`.trim(),
    );
  }
  return result.data;
}

import pg from "pg";

export type Pool = pg.Pool;

export const createPool = (connectionString: string): Pool => new pg.Pool({ connectionString });

// ponytail: one row per worker process, never pruned. LP-005 replaces this with leased jobs.
export async function beat(pool: Pool, workerId: string): Promise<void> {
  await pool.query(
    `insert into worker_heartbeat (worker_id, beat_at) values ($1, now())
     on conflict (worker_id) do update set beat_at = now()`,
    [workerId],
  );
}

export async function heartbeatAgeSeconds(pool: Pool): Promise<number | null> {
  const { rows } = await pool.query(
    "select extract(epoch from now() - max(beat_at))::float8 as age from worker_heartbeat",
  );
  return rows[0]?.age ?? null;
}
```

`packages/db/src/migrate.ts`:

```ts
import { readdir, readFile } from "node:fs/promises";
import { fileURLToPath } from "node:url";
import type { Pool } from "./index";

const DIR = fileURLToPath(new URL("../migrations/", import.meta.url));
const LOCK_KEY = 4771001;

export async function migrate(pool: Pool): Promise<string[]> {
  const client = await pool.connect();
  try {
    // Session lock: a second migrator waits here, then finds nothing left to apply.
    await client.query("select pg_advisory_lock($1)", [LOCK_KEY]);
    await client.query(
      "create table if not exists schema_migrations (name text primary key, applied_at timestamptz not null default now())",
    );
    const done = new Set(
      (await client.query("select name from schema_migrations")).rows.map((r) => r.name as string),
    );
    const applied: string[] = [];
    for (const file of (await readdir(DIR)).filter((f) => f.endsWith(".sql")).sort()) {
      if (done.has(file)) continue;
      const sql = await readFile(DIR + file, "utf8");
      await client.query("begin");
      try {
        await client.query(sql);
        await client.query("insert into schema_migrations (name) values ($1)", [file]);
        await client.query("commit");
      } catch (error) {
        await client.query("rollback");
        throw error;
      }
      applied.push(file);
    }
    return applied;
  } finally {
    await client.query("select pg_advisory_unlock($1)", [LOCK_KEY]).catch(() => {});
    client.release();
  }
}
```

import { heartbeatAgeSeconds, type Pool } from "@lp/db";

export type Health = {
  ready: boolean;
  db: "ok" | "unavailable";
  worker: "ok" | "not_seen";
  workerHeartbeatAgeSeconds: number | null;
};

const WORKER_STALE_SECONDS = 30;

export async function getHealth(pool: Pool): Promise<Health> {
  try {
    const age = await heartbeatAgeSeconds(pool);
    return {
      ready: true,
      db: "ok",
      worker: age !== null && age < WORKER_STALE_SECONDS ? "ok" : "not_seen",
      workerHeartbeatAgeSeconds: age,
    };
  } catch {
    return { ready: false, db: "unavailable", worker: "not_seen", workerHeartbeatAgeSeconds: null };
  }
}
